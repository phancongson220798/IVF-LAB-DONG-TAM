Đồng bộ lên Google Sheet

Dán đường liên kết Web App của Google Apps Script vào ô bên dưới. Dữ liệu sẽ được ghi vào 2 trang tính riêng: "BaoTri" và "ThayLoc". Nếu chưa có, làm theo hướng dẫn 4 bước phía dưới (chỉ cần làm 1 lần).

Hướng dẫn tạo Web App (4 bước)
  1. Tạo một Google Sheet mới → vào menu Tiện ích mở rộng (Extensions) → Apps Script.
  2. Xoá code mẫu, dán đoạn code bên dưới vào, rồi bấm Lưu.
  3. Bấm Triển khai (Deploy) → Bản triển khai mới (New deployment) → chọn loại Web app. Mục "Execute as" chọn Me, mục "Who has access" chọn Anyone → bấm Deploy và cho phép quyền truy cập.
  4. Copy đường liên kết Web App (kết thúc bằng /exec) và dán vào ô ở trên.
function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var sheetMaint = ss.getSheetByName('BaoTri') || ss.insertSheet('BaoTri'); var sheetFilter = ss.getSheetByName('ThayLoc') || ss.insertSheet('ThayLoc'); sheetMaint.clearContents(); sheetMaint.appendRow(['Tên thiết bị','Vị trí','Hạng mục bảo trì','Người thực hiện','Chu kỳ', 'Lần gần nhất','Hạn tiếp theo','Trạng thái','Ghi chú']); sheetFilter.clearContents(); sheetFilter.appendRow(['Loại lọc','Người thực hiện','Chu kỳ', 'Lần gần nhất','Hạn tiếp theo','Trạng thái','Ghi chú']); data.items.forEach(function(item){ if (item.category === 'filter') { sheetFilter.appendRow([item.type, item.performedBy, item.intervalText, item.lastDate, item.nextDue, item.statusText, item.notes]); } else { sheetMaint.appendRow([item.name, item.location, item.type, item.performedBy, item.intervalText, item.lastDate, item.nextDue, item.statusText, item.notes]); } }); return ContentService.createTextOutput(JSON.stringify({result:'success', count:data.items.length})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService.createTextOutput(JSON.stringify({result:'ok'})) .setMimeType(ContentService.MimeType.JSON); }
